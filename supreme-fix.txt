function loadHerbProfile(herbName){
  const profileSection = document.getElementById('herb-profile');
  if(!profileSection) return;
  
  const profileLoading = document.getElementById('profile-loading');
  const profileContent = document.getElementById('profile-content');
  
  profileSection.style.display = 'block';
  if(profileLoading) profileLoading.style.display = 'block';
  if(profileContent) profileContent.style.display = 'none';

  const herbKey = herbName.toLowerCase().trim();
  
  // Try API first, fallback to sample data
  const url = '/.netlify/functions/herb-profile';
  const payload = { herbName: herbKey };

  fetch(url, {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  })
  .then(res => {
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    return res.json();
  })
  .then(profile => {
    renderProfile(profile);
    if(profileLoading) profileLoading.style.display = 'none';
    if(profileContent) profileContent.style.display = 'block';
  })
  .catch(err => {
    console.error('API error, trying fallback:', err.message);
    const fallback = FALLBACK_PROFILES[herbKey];
    if(fallback) {
      renderProfile(fallback);
      if(profileLoading) profileLoading.innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);font-size:11px;padding:1rem;font-style:italic;">Sample profile (educational reference)</p>';
      if(profileContent) profileContent.style.display = 'block';
    } else {
      if(profileLoading) profileLoading.innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);font-style:italic;padding:2rem;">Unable to load herb profile</p>';
    }
  });
}

function renderProfile(p){
  if(!p || !p.name) return;
  
  document.getElementById('ph-cat').textContent = p.category || '—';
  document.getElementById('ph-name').textContent = p.name || '—';
  document.getElementById('ph-latin').textContent = p.latin || '—';
  document.getElementById('ph-origin').textContent = (p.origin || 'Unknown origin');
  document.getElementById('ph-category').textContent = p.category || '—';
  document.getElementById('ph-tradition').textContent = p.tradition || '—';
  document.getElementById('ph-preps').textContent = (p.preparation && typeof p.preparation === 'object' ? Object.keys(p.preparation).filter(k => p.preparation[k]).join(', ') : 'Various');
  document.getElementById('ph-safety').textContent = p.safetyLevel || 'Consult practitioner';

  document.getElementById('profile-main-content').innerHTML = p.functionalOverview ? `<div class="content-section"><h2>Overview</h2><p>${p.functionalOverview}</p></div>` : '';
  document.getElementById('profile-sidebar-content').innerHTML = '<div class="rank-display"><div class="rd-number">✓</div><div class="rd-label">Profile Loaded</div></div>';

  // Add anatomy button if it doesn't exist
  const phOriginEl = document.getElementById('ph-origin');
  if(phOriginEl && p.name && !phOriginEl.parentElement.querySelector('.anatomy-btn')){
    const btn = document.createElement('button');
    btn.className = 'anatomy-btn';
    btn.textContent = '🫀 View body systems';
    btn.style.cssText = 'background:var(--gold);color:var(--forest);border:none;padding:.5rem 1rem;border-radius:4px;font-size:13px;font-weight:500;cursor:pointer;margin-top:.75rem;transition:background .2s;font-family:"DM Sans",sans-serif;';
    btn.onmouseover = () => btn.style.background = 'var(--gold-light)';
    btn.onmouseout = () => btn.style.background = 'var(--gold)';
    btn.onclick = () => openAnatomyModal(p.name);
    phOriginEl.parentElement.insertBefore(btn, phOriginEl.nextSibling);
  }
}
