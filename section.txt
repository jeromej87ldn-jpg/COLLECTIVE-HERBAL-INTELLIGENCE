
function loadHerbProfile(herbName){
  document.getElementById('profile-bc-name').textContent = herbName;

  // Check Supabase cache first, then generate
  const url = '/.netlify/functions/herb-profile';
  const payload = { herbName: herbName.toLowerCase().trim() };

  fetch(url, {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  })
  .then(res => res.json())
  .then(profile => {
    renderProfile(profile);
    document.getElementById('profile-loading').style.display = 'none';
    document.getElementById('profile-content').style.display = 'block';
  })
  .catch(err => {
    console.error('Profile fetch error:', err);
    document.getElementById('profile-loading').innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);font-style:italic;padding:2rem;">Unable to load herb profile</p>';
  });
}

const _originalRenderProfile = window.renderProfile || function(){};
window.renderProfile = function(p){
  if(_originalRenderProfile.call) _originalRenderProfile.call(this, p);

  const phOriginEl = document.getElementById('ph-origin');
  if(phOriginEl && p.name){
    if(!phOriginEl.parentElement.querySelector('.anatomy-btn')){
      const btn = document.createElement('button');
      btn.className = 'anatomy-btn';
      btn.textContent = '🫀 View body systems';
      btn.style.cssText = 'background:var(--gold);color:var(--forest);border:none;padding:.5rem 1rem;border-radius:4px;font-size:13px;font-weight:500;cursor:pointer;margin-top:.75rem;transition:background .2s;font-family:"DM Sans",sans-serif;';
      btn.onmouseover = () => btn.style.background = 'var(--gold-light)';
      btn.onmouseout = () => btn.style.background = 'var(--gold)';
      btn.onclick = () => openAnatomyModal(p.name);
      phOriginEl.parentElement.insertBefore(btn, phOriginEl.nextSibling);
    }
  }
};

function renderProfile(p){
  document.getElementById('ph-cat').textContent = p.category || '—';
  document.getElementById('ph-name').textContent = p.name || '—';
  document.getElementById('ph-latin').textContent = p.latin || '—';
  document.getElementById('ph-origin').textContent = (p.origin || 'Unknown origin');
  document.getElementById('ph-category').textContent = p.category || '—';
  document.getElementById('ph-tradition').textContent = p.tradition || '—';
  document.getElementById('ph-preps').textContent = (p.preparation && p.preparation.length > 0 ? p.preparation[0] : 'Various');
  document.getElementById('ph-safety').textContent = p.safetyLevel || 'Consult practitioner';

  document.getElementById('profile-main-content').innerHTML = p.functionalOverview ? `<div class="content-section"><h2>Overview</h2><p>${p.functionalOverview}</p></div>` : '';
  document.getElementById('profile-sidebar-content').innerHTML = '<div class="rank-display"><div class="rd-number">✓</div><div class="rd-label">Profile Loaded</div></div>';

  _originalRenderProfile(p);
}

function toggleMobileMenu(){

// ── ANATOMY FEATURE ──
function openAnatomyModal(herbName){
  const modal = document.getElementById('anatomy-modal-overlay');
  const bodyEl = document.getElementById('anatomy-modal-body');
  const nameEl = document.getElementById('anatomy-herb-name');

  if(!modal || !bodyEl || !nameEl) return;

  nameEl.textContent = herbName;
  modal.classList.add('show');

  try{
    if(typeof ANATOMY_DATA === 'undefined'){
      bodyEl.innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);font-style:italic;">Anatomy data not loaded</p>';
      return;
    }

    const herbLc = herbName.toLowerCase().trim();
    const herbData = ANATOMY_DATA.herbSupport[herbLc];

    if(!herbData){
      bodyEl.innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);padding:1.5rem;font-style:italic;">No body system data for this herb yet</p>';
      return;
    }

    let html = '<div style="padding:0.5rem 0;">';

    if(herbData.primary && herbData.primary.length > 0){
      html += '<div style="margin-bottom:1.5rem;">';
      html += '<div style="font-size:11px;letter-spacing:.1em;text-transform:uppercase;color:var(--gold);font-weight:600;margin-bottom:.75rem;">Primary organs</div>';

      herbData.primary.forEach(organName => {
        const organ = ANATOMY_DATA.organs[organName.toLowerCase()];
        if(organ){
          html += '<div style="background:var(--parchment-dark);border:1px solid var(--parchment-deep);border-radius:6px;padding:1rem;margin-bottom:.75rem;">';
          html += '<div style="font-family:\'Cormorant Garamond\',serif;font-size:16px;font-weight:500;color:var(--charcoal);margin-bottom:.25rem;">' + organ.name + '</div>';
          html += '<div style="font-size:12px;color:var(--charcoal-mid);font-style:italic;margin-bottom:.6rem;">' + organ.latin + '</div>';
          html += '<div style="font-size:13px;color:var(--charcoal-mid);line-height:1.6;">' + organ.functions + '</div>';
          html += '</div>';
        }
      });
      html += '</div>';
    }

    if(herbData.secondary && herbData.secondary.length > 0){
      html += '<div style="margin-bottom:1rem;">';
      html += '<div style="font-size:11px;letter-spacing:.1em;text-transform:uppercase;color:var(--charcoal-mid);font-weight:600;margin-bottom:.75rem;">Supporting organs</div>';
      html += '<div style="display:grid;grid-template-columns:1fr 1fr;gap:.75rem;">';

      herbData.secondary.forEach(organName => {
        const organ = ANATOMY_DATA.organs[organName.toLowerCase()];
        if(organ){
          html += '<div style="background:rgba(200,149,42,.06);border:1px solid rgba(200,149,42,.15);border-radius:4px;padding:.75rem;">';
          html += '<div style="font-weight:500;color:var(--charcoal);font-size:13px;margin-bottom:.25rem;">' + organ.name + '</div>';
          html += '<div style="font-size:11px;color:var(--charcoal-mid);font-style:italic;">' + organ.latin + '</div>';
          html += '</div>';
        }
      });
      html += '</div>';
      html += '</div>';
    }

    html += '<div style="background:rgba(26,58,42,.07);border:1px solid rgba(26,58,42,.15);border-radius:6px;padding:1rem;margin-top:1.5rem;">';
    html += '<div style="font-size:12px;color:var(--charcoal-mid);line-height:1.7;">';
    html += '<strong style="color:var(--charcoal);">About body affinity:</strong> This mapping is based on traditional use patterns and modern pharmacological research. Primary organs have 2+ keyword matches in the herb\'s profile; supporting organs have 1 match.';
    html += '</div>';
    html += '</div>';
    html += '</div>';

    bodyEl.innerHTML = html;
  }catch(e){
    console.error('Anatomy modal error:', e);
    bodyEl.innerHTML = '<p style="text-align:center;color:var(--charcoal-mid);font-style:italic;padding:1.5rem;">Unable to load anatomy data</p>';
  }
}

function closeAnatomyModal(){
  const modal = document.getElementById('anatomy-modal-overlay');
  if(modal) modal.classList.remove('show');
}
